1 BlockHeader Extension (pkg/utxo/block.go)

    Version       uint32
    PrevBlockHash [32]byte
    MerkleRoot    [32]byte
    Timestamp     uint64
    Height        uint64
    Proposer      [32]byte

    // New PoAIW fields (Version >= 2)
    InferencePoW  []byte  // Inference proof of work

2 Validation Logic Extension (pkg/utxo/validator.go) 
func (v *ChainBlockValidator) ValidateBlock(block *Block) error {
    // Existing validation...
    if err := v.validateExisting(block); err != nil {
        return err
    }

    // PoAIW verification (Version >= 2)
    if block.Header.Version >= 2 {
        return v.validatePoAIW(block)
    }
    return nil
}

func (v *ChainBlockValidator) validatePoAIW(block *Block) error {
    // 1. PoW verification
    if !verifyInferencePoW(block) {
        return fmt.Errorf("inference PoW invalid")
    }
    // 2. Staking verification
    if !verifyMinStake(block) {
        return fmt.Errorf("insufficient stake")
    }
    return nil
}

6. Recommended Upgrade Plan: Testnet First

Phase 1: Preparation (Week 1-2)

  • Deploy PoAIW code on testnet
  • Run integration tests
  • Update documentation and tooling

Phase 2: Testnet Validation (Week 3-4)

  • Enable PoAIW on testnet (Version 2 blocks)
  • Monitor network stability
  • Collect node feedback

Phase 3: Mainnet Proposal (Week 5-6)

  • Publish AIP (AIB Improvement Proposal)
  • Community discussion and voting
  • Set activation height

Phase 4: Mainnet Upgrade (Week 7+)

  • Activate PoAIW at the designated height
  • Watch for chain splits
  • Emergency response plan

7. Key Decision Points

⚠️ Questions Requiring Community Decisions

8. Conclusion

The current testnet can be upgraded to simplified PoAIW

However, this requires a hard fork; a gradual testnet-first approach is recommended:

  • ✅ Code is ready (tests passing)
  • ✅ Architecture supports versioned upgrades
  • ⚠️ Requires community consensus
  • ⚠️ Requires a cautious activation strategy
